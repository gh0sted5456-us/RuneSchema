cmake_minimum_required(VERSION 3.22)
project(Helpy VERSION 2.0.4 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Helpy requires the Windows x64 MSVC toolchain")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_compile_options(/utf-8 /Zc:preprocessor /EHsc /permissive- /Z7 /experimental:deterministic)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS IMGUI_IMPL_WIN32_DISABLE_GAMEPAD)
include_directories(source/include vendor/imgui vendor/json/include)
add_library(helpy_imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp
  vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
  vendor/imgui/backends/imgui_impl_win32.cpp vendor/imgui/backends/imgui_impl_dx11.cpp)
set(HELPY_UI_SOURCES source/src/Generator/HelpyStandaloneWindow.cpp
  source/src/Generator/ToolRequestState.cpp source/src/Core/ConfigFiles.cpp source/src/HelpyPaths.cpp)
add_library(RuneSchemaHelpyPlugin SHARED source/plugins/helpy/HelpyPlugin.cpp ${HELPY_UI_SOURCES})
target_compile_definitions(RuneSchemaHelpyPlugin PRIVATE RUNESCHEMA_PLUGIN_CLIENT=1)
target_link_libraries(RuneSchemaHelpyPlugin PRIVATE helpy_imgui d3d11 dxgi d3dcompiler dwmapi shell32 windowscodecs ole32)
set_target_properties(RuneSchemaHelpyPlugin PROPERTIES OUTPUT_NAME RuneSchema.Helpy RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_link_options(RuneSchemaHelpyPlugin PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO /PDBALTPATH:RuneSchema.Helpy.pdb)

option(HELPY_BUILD_TESTS "Build standalone Helpy contract and rendering tests" ${PROJECT_IS_TOP_LEVEL})
if(HELPY_BUILD_TESTS)
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-temp")
foreach(test item-authoring map-landing-search source-cache item-paging item-bay-layout item-cart item-transfer item-presentation map-projection map-bridge helpy-kit-files recipe-workspace authoring-policy helpy-lazy-thread)
  add_executable(${test} source/tests/${test}.cpp source/src/Core/ConfigFiles.cpp)
  # Assertions are meaningful checks in a Release build, too.
  target_compile_options(${test} PRIVATE /UNDEBUG)
  add_test(NAME ${test} COMMAND ${test})
  set_tests_properties(${test} PROPERTIES ENVIRONMENT "TEMP=${CMAKE_CURRENT_BINARY_DIR}/test-temp;TMP=${CMAKE_CURRENT_BINARY_DIR}/test-temp")
endforeach()
target_sources(helpy-lazy-thread PRIVATE source/src/Generator/ToolRequestState.cpp)
target_sources(map-bridge PRIVATE ../raw/src/Generator/ToolRequestState.cpp)
target_include_directories(map-bridge BEFORE PRIVATE ../raw/include)
target_include_directories(map-landing-search BEFORE PRIVATE ../raw/include)
add_executable(source-cache-core source/tests/source-cache.cpp)
target_include_directories(source-cache-core BEFORE PRIVATE ../raw/include)
target_compile_options(source-cache-core PRIVATE /UNDEBUG)
add_test(NAME source-cache-core COMMAND source-cache-core)
set_tests_properties(source-cache-core PROPERTIES ENVIRONMENT "TEMP=${CMAKE_CURRENT_BINARY_DIR}/test-temp;TMP=${CMAKE_CURRENT_BINARY_DIR}/test-temp")
add_executable(icon-demand source/tests/icon-demand.cpp source/src/Generator/ToolRequestState.cpp source/src/HelpyPaths.cpp)
target_compile_options(icon-demand PRIVATE /UNDEBUG)
target_link_libraries(icon-demand PRIVATE helpy_imgui)
add_test(NAME icon-demand COMMAND icon-demand)
set_tests_properties(icon-demand PROPERTIES ENVIRONMENT "LOCALAPPDATA=${CMAKE_CURRENT_BINARY_DIR}/test-user")
add_test(NAME helpy-source-cache-parity COMMAND "${CMAKE_COMMAND}" -E compare_files
  "${CMAKE_CURRENT_SOURCE_DIR}/source/include/Core/HelpySourceCache.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/../raw/include/Core/HelpySourceCache.h")
add_executable(helpy-popup-input source/tests/helpy-popup-input.cpp)
add_test(NAME helpy-popup-input COMMAND helpy-popup-input "${CMAKE_CURRENT_SOURCE_DIR}/source/src/Generator/HelpyStandaloneWindow.cpp")
add_executable(helpy-navigation source/tests/helpy-navigation.cpp)
add_test(NAME helpy-navigation COMMAND helpy-navigation)
add_executable(helpy-plugin-contract source/tests/helpy-plugin-contract.cpp)
target_compile_definitions(helpy-plugin-contract PRIVATE HELPY_PLUGIN_PACKAGE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../plugins/RuneSchema.Helpy")
add_test(NAME helpy-plugin-contract COMMAND helpy-plugin-contract "$<TARGET_FILE:RuneSchemaHelpyPlugin>")
add_test(NAME helpy-server-desktop COMMAND helpy-plugin-contract "$<TARGET_FILE:RuneSchemaHelpyPlugin>" --server-desktop)
set_tests_properties(helpy-server-desktop PROPERTIES TIMEOUT 30 ENVIRONMENT "LOCALAPPDATA=${CMAKE_CURRENT_BINARY_DIR}/test-server-user")
add_executable(helpy-transport source/tests/helpy-transport.cpp ${HELPY_UI_SOURCES})
target_compile_definitions(helpy-transport PRIVATE RUNESCHEMA_PLUGIN_CLIENT=1)
target_compile_options(helpy-transport PRIVATE /UNDEBUG)
target_link_libraries(helpy-transport PRIVATE helpy_imgui d3d11 dxgi d3dcompiler dwmapi shell32 windowscodecs ole32)
add_test(NAME helpy-transport COMMAND helpy-transport)

# Same production renderer with a synthetic host; never shipped as the plugin.
add_executable(HelpyPreview WIN32 source/tests/HelpyPreview.cpp ${HELPY_UI_SOURCES})
target_compile_definitions(HelpyPreview PRIVATE HELPY_PREVIEW=1)
target_link_libraries(HelpyPreview PRIVATE helpy_imgui d3d11 dxgi d3dcompiler dwmapi shell32 windowscodecs ole32)
add_test(NAME helpy-render-workspaces COMMAND HelpyPreview --capture "${CMAKE_CURRENT_BINARY_DIR}/preview" --assets "${CMAKE_CURRENT_SOURCE_DIR}/../plugins/RuneSchema.Helpy")
set_tests_properties(helpy-render-workspaces PROPERTIES ENVIRONMENT "LOCALAPPDATA=${CMAKE_CURRENT_BINARY_DIR}/test-user")
add_test(NAME helpy-render-compact COMMAND HelpyPreview --capture "${CMAKE_CURRENT_BINARY_DIR}/preview-compact" --assets "${CMAKE_CURRENT_SOURCE_DIR}/../plugins/RuneSchema.Helpy" --size 960 700)
add_test(NAME helpy-render-minimum COMMAND HelpyPreview --capture "${CMAKE_CURRENT_BINARY_DIR}/preview-minimum" --assets "${CMAKE_CURRENT_SOURCE_DIR}/../plugins/RuneSchema.Helpy" --size 760 560)
set_tests_properties(helpy-render-compact helpy-render-minimum PROPERTIES ENVIRONMENT "LOCALAPPDATA=${CMAKE_CURRENT_BINARY_DIR}/test-user")
endif()
